            <div class="flex-grow space-y-6">
                <div class="bg-slate-950 p-6 rounded-2xl border border-slate-800 h-64 overflow-y-auto">
                    <div id="doc-content" class="text-slate-400">
                        <p class="mb-4">Once you stop speaking, AI will structure your query into:</p>
                        <ul class="list-disc ml-5 space-y-2 text-sm">
                            <li>Technical Summary</li>
                            <li>Action Items</li>
                            <li>Automation Triggers</li>
                        </ul>
                    </div>
                </div>

                <div class="p-4 bg-blue-500/10 border border-blue-500/20 rounded-xl flex items-center justify-between">
                    <div>
                        <p class="text-xs font-bold text-blue-400 uppercase">System Status</p>
                        <p id="automation-status" class="text-sm text-slate-300 italic">Awaiting input...</p>
                    </div>
                    <i class="fas fa-robot text-blue-500 text-2xl"></i>
                </div>

                <button class="w-full py-4 bg-blue-600 hover:bg-blue-500 text-white font-bold rounded-xl transition-all shadow-lg shadow-blue-900/20">
                    Push to Production
                </button>
            </div>
        </div>
    </div>

    <script>
        const startBtn = document.getElementById('start-btn');
        const micIcon = document.getElementById('mic-icon');
        const statusText = document.getElementById('status-text');
        const transcriptBox = document.getElementById('transcript-box');
        const docContent = document.getElementById('doc-content');
        const autoStatus = document.getElementById('automation-status');

        // Initialize Web Speech API
        const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
        
        if (!SpeechRecognition) {
            statusText.innerText = "Browser does not support Speech API.";
        } else {
            const recognition = new SpeechRecognition();
            recognition.continuous = true;
            recognition.interimResults = true;

            let isRecording = false;

            startBtn.onclick = () => {
                if (!isRecording) {
                    recognition.start();
                } else {
                    recognition.stop();
                }
            };

            recognition.onstart = () => {
                isRecording = true;
                startBtn.classList.add('pulse', 'bg-red-500');
                statusText.innerText = "Listening... Speak now";
                transcriptBox.innerText = "";
            };

            recognition.onresult = (event) => {
                let interimTranscript = '';
                for (let i = event.resultIndex; i < event.results.length; ++i) {
                    if (event.results[i].isFinal) {
                        processFinalText(event.results[i][0].transcript);
                    } else {
                        interimTranscript += event.results[i][0].transcript;
                    }
                }
                transcriptBox.innerText = interimTranscript;
            };

            recognition.onend = () => {
                isRecording = false;
                startBtn.classList.remove('pulse', 'bg-red-500');
                statusText.innerText = "Processing Documentation...";
            };

            function processFinalText(text) {
                transcriptBox.innerText = text;
                // Mocking the AI "Cleaning Up" the text
                docContent.innerHTML = `
                    <h3 class="text-blue-400 font-bold mb-2 uppercase text-xs">Generated Summary</h3>
                    <p class="text-slate-200 mb-4">"The user requested: ${text}"</p>
                    <h3 class="text-blue-400 font-bold mb-2 uppercase text-xs">Next Steps</h3>
                    <ul class="list-disc ml-4 text-sm text-slate-300">
                        <li>Log entry into Documentation Hub</li>
                        <li>Notify engineering team via Webhook</li>
                    </ul>
                `;
                autoStatus.innerText = "Triggering Zapier/Jira Integration...";
            }
        }
    </script>
